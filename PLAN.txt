### 4.2 Service (`src/services/productor.service.ts`)
```ts
async crearConPredio(data: CrearProductorDTO) {
  // 1. Pre-validar parroquiaId existe
  const parroquia = await prisma.parroquia.findUnique({ where: { id: data.finca.parroquiaId } })
  if (!parroquia) throw new BadRequestError('Parroquia no encontrada')

  // 2. Transacción atómica
  return prisma.$transaction(async (tx) => {
    const productor = await tx.productor.create({
      data: {
        cedulaRif: data.cedulaRif,
        nombre: data.nombre,
        apellido: data.apellido,
        telefono: data.telefono,
        predios: {
          create: {
            nombre: data.finca.nombre,
            sector: data.finca.sector,
            latitud: data.finca.latitud,
            longitud: data.finca.longitud,
            parroquiaId: data.finca.parroquiaId
          }
        }
      }
    })
    return { productorId: productor.id }
  })
}
```

### 5.3 Utilidad Hamming (`src/utils/hash-comparer.util.ts`)
```ts
export function hammingDistance(hash1: string, hash2: string): number {
  // Ambos hex strings de 16 chars (64 bits)
  const buf1 = Buffer.from(hash1, 'hex')
  const buf2 = Buffer.from(hash2, 'hex')
  let diff = 0
  for (let i = 0; i < buf1.length; i++) {
    diff += (buf1[i] ^ buf2[i]).toString(2).split('1').length - 1
  }
  return diff
}

export function esDuplicado(nuevoHash: string, hashesExistentes: string[]): boolean {
  return hashesExistentes.some(h => hammingDistance(nuevoHash, h) <= 6)
}
```

### 5.4 Service Imagen (`src/services/imagen.service.ts`)
```ts
import { imageHash } from 'image-hash'
import sharp from 'sharp'

async procesarYValidar(buffer: Buffer): Promise<string> {
  // 1. Normalizar con Sharp
  const normalized = await sharp(buffer)
    .resize(256, 256, { fit: 'fill' })
    .grayscale()
    .normalize()
    .png()
    .toBuffer()

  // 2. pHash
  const hash = await new Promise<string>((resolve, reject) => {
    imageHash.phash(normalized, 64, (err, hash) => err ? reject(err) : resolve(hash))
  })

  // 3. Comparar contra BD
  const existentes = await prisma.marca.findMany({ select: { hashImagen: true } })
  if (esDuplicado(hash, existentes.map(m => m.hashImagen))) {
    throw new ConflictError('Estructura de marca rechazada. Se detectó una coincidencia de patrón superior al límite tolerado con una marca registrada previamente.')
  }

  // 4. Guardar archivo físico
  const filename = `${hash}.png`
  const filepath = path.join('storage/marcas', filename)
  await fs.promises.writeFile(filepath, normalized)

  return hash  // Se guarda en BD
}
```

## 📊 FASE 6: Reportes (Protegido)

### 6.1 Service (`src/services/reporte.service.ts`)
```ts
async listarProductores(filtros: { activo?: boolean; estadoId?: number; municipioId?: number }, page: number, limit: number) {
  const where: Prisma.ProductorWhereInput = {}
  
  if (filtros.activo !== undefined) where.activo = filtros.activo
  if (filtros.estadoId) {
    where.predios = { some: { parroquia: { municipio: { estadoId: filtros.estadoId } } } }
  }
  if (filtros.municipioId) {
    where.predios = { some: { parroquia: { municipioId: filtros.municipioId } } }
  }

  const [data, total] = await Promise.all([
    prisma.productor.findMany({
      where,
      skip: (page - 1) * limit,
      take: limit,
      orderBy: { createdAt: 'desc' },
      select: {
        id: true, cedulaRif: true, nombre: true, apellido: true, activo: true,
        _count: { select: { predios: true } },
        predios: { take: 1, orderBy: { id: 'asc' }, select: { parroquia: { select: { municipio: { select: { nombre: true, estado: { select: { nombre: true } } } } } } } },
        marca: { select: { id: true } }
      }
    }),
    prisma.productor.count({ where })
  ])

  return {
    totalResultados: total,
    paginaActual: page,
    paginasTotales: Math.ceil(total / limit),
    data: data.map(p => ({
      id: p.id,
      cedulaRif: p.cedulaRif,
      nombreCompleto: `${p.nombre} ${p.apellido}`,
      activo: p.activo,
      totalFincas: p._count.predios,
      ubicacionPrincipal: p.predios[0] 
        ? `${p.predios[0].parroquia.municipio.estado.nombre}, ${p.predios[0].parroquia.municipio.nombre}`
        : 'Sin predios',
      tieneMarcaRegistrada: !!p.marca
    }))
  }
}
```
